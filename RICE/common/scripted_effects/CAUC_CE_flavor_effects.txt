
CAUC_establish_transcaucasia_effect = {

	title:d_shirvan = { set_de_jure_liege_title = title:k_shirvan }
 title:k_georgia = { set_de_jure_liege_title = title:e_CAUC_transcaucasia }
 title:k_armenia = { set_de_jure_liege_title = title:e_CAUC_transcaucasia }
 title:k_shirvan = { set_de_jure_liege_title = title:e_CAUC_transcaucasia }
 create_title_and_vassal_change = {
		type = created
		save_scope_as = change
	}

	title:e_CAUC_transcaucasia = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}

	resolve_title_and_vassal_change = scope:change

	# Prestige
	add_prestige = 750

	# Also gain a large amount of dynasty prestige.
	dynasty = { add_dynasty_prestige = 500 }

	# Add Nickname
	set_nickname_effect = { NICKNAME = nick_the_great }

	# Byzantine Emperor hates new pretender & claims the lands in question
	if = {
		limit = {
			exists = title:e_byzantium.holder
		}
		title:e_byzantium.holder = {
			add_opinion = {
				target = root
				modifier = CAUC_imperial_pretender
			}
		}
	}

	# Persian Emperor hates new pretender & claims the lands in question
	if = {
		limit = {
			exists = title:e_persia.holder
		}
		title:e_persia.holder = {
			add_opinion = {
				target = root
				modifier = CAUC_imperial_pretender
			}
		}
	}

}
