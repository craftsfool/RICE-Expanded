# Cast is stored on the ruler and recovered at every delayed stage, including after reload.
CAUC_rus_speaker_effect = {
 if = { limit = { has_variable = CAUC_rus_speaker var:CAUC_rus_speaker = { is_alive = yes } }
  var:CAUC_rus_speaker = { save_scope_as = CAUC_rus_speaker }
 } else = {
  create_character = { age = 30 dynasty = none gender_female_chance = 0 random_traits = no
   location = root.capital_province culture = culture:russian faith = faith:slavic_pagan
   trait = education_martial_4 trait = brave trait = gallant trait = lifestyle_blademaster martial = 26 prowess = 32 diplomacy = 12 learning = 8 save_scope_as = CAUC_rus_speaker
  }
  add_visiting_courtier = scope:CAUC_rus_speaker
  scope:CAUC_rus_speaker = { add_trait_xp = { trait = lifestyle_blademaster value = 100 } }
  set_variable = { name = CAUC_rus_speaker value = scope:CAUC_rus_speaker }
 }
}
