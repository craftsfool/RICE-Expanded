# Follows the native 1.20 historical-heresy pattern: named founder once,
# creation in the founder's character scope, anonymous speakers for resurgences.
# Zarehavantsi is a geographic designation, not an attested hereditary dynasty.
CAUC_tondrakian_prepare_inquiry_effect = {
 if = {
  limit = { NOT = { exists = rite:CAUC_tondrakian_rite } }
  create_character = {
   age = 40
   name = CAUC_smbat_zarehavantsi_name
   dynasty = none
   gender_female_chance = 0
   random_traits = no
   location = title:c_apahunik.title_province
   culture = culture:armenian
   faith = faith:armenian_apostolic
   trait = education_learning_4
   trait = zealous
   trait = brave
   trait = theologian
   diplomacy = 12
   martial = 5
   stewardship = 10
   intrigue = 6
   learning = 20
   prowess = 5
   save_scope_as = CAUC_tondrakian_speaker
  }
  scope:CAUC_tondrakian_speaker = {
   save_scope_as = historical_character
   historical_character_finalization_effect = yes
   set_variable = CAUC_smbat_zarehavantsi
   set_global_variable = { name = CAUC_tondrakian_founder value = this }
   set_global_variable = { name = CAUC_tondrakian_origin value = title:c_apahunik }
   set_global_variable = { name = CAUC_tondrakian_founding_year value = current_year }
   create_rite_from_type = { type = CAUC_tondrakian_rite save_scope_as = new_rite convert = yes }
   add_trait = heresiarch
   add_character_flag = ai_will_not_convert
  }
 }
 CAUC_tondrakian_select_speaker_effect = yes
}

# Native C3 geographic seed, without C3's later creation of a separate faith.
# Once only: reloads must not undo a ruler's subsequent suppression/conversion.
CAUC_tondrakian_seed_counties_effect = {
 if = {
  limit = { exists = rite:CAUC_tondrakian_rite NOT = { has_global_variable = CAUC_tondrakian_counties_seeded } }
  title:c_apahunik = {
   if = {
    limit = { faith = faith:armenian_apostolic NOT = { has_variable = CAUC_tondrakian_inquiry_choice var:CAUC_tondrakian_inquiry_choice = 5 } }
    set_county_rite = rite:CAUC_tondrakian_rite
    every_neighboring_county = {
     limit = { faith = faith:armenian_apostolic holder.faith = faith:armenian_apostolic }
     set_county_rite = rite:CAUC_tondrakian_rite
    }
   }
  }
  rite:CAUC_tondrakian_rite = {
   set_variable = { name = pam_rite_grace_period value = yes days = rite_growth_grace_period_days }
   set_variable = { name = pam_rite_founding_window value = yes years = 1 }
  }
  set_global_variable = { name = CAUC_tondrakian_counties_seeded value = yes }
 }
}

# Required saved scopes are identical to faith_creation.1020/1021. The native
# 1020 dispatcher is player-only, so dispatch 1021 for this historical AI founder.
CAUC_tondrakian_notification_scopes_effect = {
 global_var:CAUC_tondrakian_founder = { save_scope_as = founder }
 faith:armenian_apostolic = {
  save_scope_as = old_faith
  main_rite = { save_scope_as = source_rite }
 }
 rite:CAUC_tondrakian_rite = { save_scope_as = new_rite }
 rite_growth_resolve_differences_effect = yes
}

# Native conversion includes matching domain counties, courtiers, conversion
# consequences and the ten-year anti-reconversion flag. It keeps parent faith.
CAUC_tondrakian_join_effect = {
 # A Christian of another denomination first joins the Armenian parent faith.
 # Convert only domain counties/courtiers sharing that former faith.
 if = {
  limit = { faith != faith:armenian_apostolic }
  faith = { save_scope_as = CAUC_before_join_faith }
  set_character_faith_with_conversion = faith:armenian_apostolic
  every_held_title = {
   limit = { tier = tier_county faith = scope:CAUC_before_join_faith }
   set_county_rite = rite:CAUC_tondrakian_rite
  }
  pam_convert_court_to_faith_effect = { OLD_FAITH = scope:CAUC_before_join_faith FAITH = faith:armenian_apostolic }
 }
 CAUC_tondrakian_notification_scopes_effect = yes
 pam_heresy_convert_ruler_to_new_rite_effect = yes
}

# The first testimony and final verdict reach every eligible Caucasian ruler.
# Only rulers who opt into the hearing receive the two intervening stages.
CAUC_tondrakian_broadcast_followup_effect = {
 title:c_apahunik.holder = { save_scope_as = CAUC_tondrakian_host }
 every_ruler = {
  limit = {
   CAUC_tondrakian_broadcast_recipient_trigger = yes
   NOT = { this = scope:CAUC_tondrakian_host }
   OR = {
    scope:CAUC_tondrakian_news_stage = 141
    scope:CAUC_tondrakian_news_stage = 144
    has_character_flag = CAUC_tondrakian_follow_testimony
   }
  }
  trigger_event = { id = CAUC.0151 days = 3 }
 }
}

CAUC_tondrakian_select_speaker_effect = {
 if = {
  limit = {
   exists = global_var:CAUC_tondrakian_founder
   global_var:CAUC_tondrakian_founder = { is_alive = yes }
  }
  global_var:CAUC_tondrakian_founder = { save_scope_as = CAUC_tondrakian_speaker }
 }
 else = {
  create_character = {
   age = 35
   dynasty = none
   gender_female_chance = 50
   random_traits = no
   location = title:c_apahunik.title_province
   culture = culture:armenian
   faith = faith:armenian_apostolic
   trait = education_learning_3
   trait = zealous
   trait = humble
   learning = 14
   save_scope_as = CAUC_tondrakian_speaker
  }
  scope:CAUC_tondrakian_speaker = {
   set_character_rite = rite:CAUC_tondrakian_rite
   add_character_flag = ai_will_not_convert
  }
 }
 set_variable = { name = CAUC_tondrakian_inquiry_speaker value = scope:CAUC_tondrakian_speaker }
}

# Restore the speaker after a save/reload or switch to a living follower if the
# current speaker died. The permanent founder reference is never overwritten.
CAUC_tondrakian_recover_speaker_effect = {
 if = {
  limit = {
   has_variable = CAUC_tondrakian_inquiry_speaker
   var:CAUC_tondrakian_inquiry_speaker = { is_alive = yes }
  }
  var:CAUC_tondrakian_inquiry_speaker = { save_scope_as = CAUC_tondrakian_speaker }
 }
 else = { CAUC_tondrakian_select_speaker_effect = yes }
}
